# Nav Shell Implementation Plan

> **For agentic workers:** Use this plan task-by-task. Steps use checkbox syntax for tracking.

**Goal:** Build the three-breakpoint responsive nav shell for #52 (Slice 5 — Nav 1.0). Connect the app's five surfaces (Agenda, Roster, Profile, Invite, Collectives) via a persistent navigation component integrated in `+layout.svelte`. Three responsive states: bottom tab bar (portrait phone), spine rail (landscape/tablet), top bar (desktop).

**Architecture:** A declarative entry list (`entries.ts`) defines nav items with visibility predicates. An admin store (`adminStore.ts`) resolves admin-ness via a single Entu call (T5.1 mechanism). The `NavShell.svelte` component wraps page content and renders the nav in three CSS-driven responsive states. The root layout wires auth state, completion gate, admin store, and collective picker mode into NavShell props.

**Tech Stack:** SvelteKit 2 / Svelte 5 (Runes) / Vitest + happy-dom / Paraglide i18n / pnpm

## Global Constraints

- Svelte 5 Runes only (`$state`, `$derived`, `$props`, `$effect`) — NEVER legacy `export let` / `$:`
- pnpm, never npm; path: `export PATH="$HOME/.npm-global/bin:$PATH"`
- No new Entu queries beyond the admin determination call
- Nav visibility is convenience, never a security boundary
- Icons are inline SVG strings using `currentColor`, no icon library
- CSS custom properties from `app.css` `@theme` for colors (`--color-paper`, `--color-ink`, etc.)

### 1.2 Create admin determination store

- [ ] Create `src/lib/nav/adminStore.ts`

```ts
// src/lib/nav/adminStore.ts
import { writable, type Writable } from 'svelte/store';
import { entuFetch } from '$lib/entu/request';
import type { EntuCfg } from '$lib/seasons/entuSeasons';

export type AdminState = 'loading' | 'admin' | 'not-admin' | 'error';

export const adminStore: Writable<AdminState> = writable('loading');

export function resetAdmin(): void {
	adminStore.set('loading');
}

export async function resolveAdmin(
	cfg: EntuCfg,
	personId: string,
	fetchImpl: typeof fetch = fetch
): Promise<AdminState> {
	try {
		const res = await entuFetch(
			cfg.db,
			'entity?_type.string=organization&props=_owner,_editor&limit=1',
			cfg.token,
			{},
			fetchImpl
		);
		if (!res.ok) return 'error';

		const body = (await res.json()) as {
			entities?: Array<{
				_id: string;
				_owner?: Array<{ reference?: string }>;
				_editor?: Array<{ reference?: string }>;
			}>;
		};
		const org = body.entities?.[0];
		if (!org) return 'not-admin';

		const isOwner = (org._owner ?? []).some((p) => p.reference === personId);
		const isEditor = (org._editor ?? []).some((p) => p.reference === personId);
		return isOwner || isEditor ? 'admin' : 'not-admin';
	} catch {
		return 'error';
	}
}
```

```svelte
<script lang="ts">
	import type { Snippet } from 'svelte';
	import type { NavEntry, NavContext } from '$lib/nav/entries';

	let {
		entries,
		activeRoute,
		completionLocked = false,
		anonymous = false,
		isAdmin = false,
		hasMultipleCollectives = false,
		children,
	}: {
		entries: NavEntry[];
		activeRoute: string;
		completionLocked?: boolean;
		anonymous?: boolean;
		isAdmin?: boolean;
		hasMultipleCollectives?: boolean;
		children: Snippet;
	} = $props();

	const ctx: NavContext = $derived({ isAdmin, hasMultipleCollectives });
	const visibleEntries = $derived(entries.filter((e) => e.visible(ctx)));

	function isActive(route: string): boolean {
		if (route === '/') return activeRoute === '/';
		return activeRoute.startsWith(route);
	}

	function handleKeydown(e: KeyboardEvent): void {
		const nav = (e.currentTarget as HTMLElement);
		const links = Array.from(nav.querySelectorAll<HTMLAnchorElement>('a:not([tabindex="-1"])'));
		const idx = links.indexOf(e.target as HTMLAnchorElement);
		if (idx < 0) return;

		let next = -1;
		if (e.key === 'ArrowRight' || e.key === 'ArrowDown') {
			next = (idx + 1) % links.length;
		} else if (e.key === 'ArrowLeft' || e.key === 'ArrowUp') {
			next = (idx - 1 + links.length) % links.length;
		}
		if (next >= 0) {
			e.preventDefault();
			links[next].focus();
		}
	}
</script>
```
